```bash
# OpenRouter API

API=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

# Бесплатные модели для ротации
CUSTOM_MODELS=-all;Free:amazon/nova-2-lite-v1:free,tngtech/tng-r1t-chimera:free,z-ai/glm-4.5-air:free,kwaipilot/kat-coder-pro:free

# Системный промпт (поведение бота)
SYSTEMPROMPT=You are a professional personal AI assistant. You help with information search, document analysis, and answering questions. Always respond in the same language as the user's question (Russian, Portuguese, English, Chinese, etc.). Be concise, helpful, and professional. You have internet access and can analyze documents.
```
